-- CREATE TABLES

CREATE TABLE customer
(
  customerid VARCHAR(5) 	NOT NULL,
  cname 	VARCHAR(70) 	NOT NULL,
  email 	VARCHAR(255) 	NOT NULL,
  street 	VARCHAR(30) 	NOT NULL,
  city 		VARCHAR(25) 	NOT NULL,
  state 	VARCHAR(15) 	NOT NULL,
  zip 		VARCHAR(10) 	NOT NULL,
  phonenumber 	VARCHAR(20) 	DEFAULT 'not provided',
  PRIMARY KEY (customerid));

CREATE TABLE employee
(
  employeeid 	VARCHAR(5) 	NOT NULL,
  role 		VARCHAR(30) 	DEFAULT 'Sales Associate',
  empname 	VARCHAR(70) 	NOT NULL,
  street 	VARCHAR(40) 	NOT NULL,
  city 		VARCHAR(20) 	NOT NULL,
  state 	VARCHAR(30) 	NOT NULL,
  zip 		VARCHAR(10) 	NOT NULL,
  phonenumber 	VARCHAR(20) 	NOT NULL,
  PRIMARY KEY (employeeid) );

-- INDEX
CREATE INDEX emp_role_index ON employee(role);
-- DROP INDEX emp_role_index;

CREATE TABLE orders
(
  orderid 	VARCHAR(6) 	NOT NULL,
  odate 	DATE 		NOT NULL,
  customerid 	VARCHAR(5) 	NOT NULL,
  employeeid 	VARCHAR(5) 	DEFAULT '1',
  PRIMARY KEY (orderid),
  FOREIGN KEY (customerid) REFERENCES customer(customerid) ON DELETE CASCADE, -- remove record if referenced customerid is deleted
  FOREIGN KEY (employeeid) REFERENCES employee(employeeid) ON DELETE SET NULL -- set field to null if employee has been removed
);


CREATE TABLE inventory
(
  inventoryid 	VARCHAR(6)	 	 NOT NULL,
  idate 	DATE 		 NOT NULL,
  purchaseprice NUMERIC(4,2)	 NOT NULL	CHECK (purchaseprice > 0), -- checks the price is a positive number
  PRIMARY KEY (inventoryid)
);

CREATE TABLE supplier 
(
  suppentryid 	VARCHAR(6) 	NOT NULL,
  supplierid 	VARCHAR(6) 	NOT NULL,
  compname 	VARCHAR(100) 	NOT NULL,
  street 	VARCHAR(30) 	NOT NULL,
  city 		VARCHAR(25) 	NOT NULL,
  state 	VARCHAR(15) 	NOT NULL,
  zip 		VARCHAR(20) 	NOT NULL,
  phonenumber 	VARCHAR(20) 	NOT NULL,
  inventoryid 	VARCHAR(6) 	NOT NULL,
  PRIMARY KEY (suppentryid),
  FOREIGN KEY (inventoryid) REFERENCES Inventory(inventoryid) ON DELETE SET NULL -- set inventoryid to null if deleted in parent table
);

CREATE TABLE customerinquiry
(
  custinqid 	VARCHAR(6) 	NOT NULL,
  customerid 	VARCHAR(5) 	NOT NULL,
  descrprob 	VARCHAR(255),
  faq 		VARCHAR(255),
  employeeid 	VARCHAR(5) 	DEFAULT '1',
  PRIMARY KEY (custinqid),
  FOREIGN KEY (customerid) REFERENCES Customer(customerid) ON DELETE CASCADE, -- remove record if referenced customerid is deleted
  FOREIGN KEY (employeeid) REFERENCES Employee(employeeid) ON DELETE SET NULL -- set employeeid to null if deleted in parent table
);

CREATE TABLE discount
(
  code 		VARCHAR(15) 	NOT NULL,
  discamount 	NUMERIC(4,2) 	NOT NULL,
  expdate 	DATE 		NOT NULL,
  PRIMARY KEY (code)
);

CREATE TABLE book
(
  bookid 	VARCHAR(6) 	NOT NULL,
  title 	VARCHAR(150) 	NOT NULL,
  isbn 		VARCHAR(20) 	NOT NULL,
  stockquantity INT 		NOT NULL	CHECK (stockquantity >= 0), -- checks there's not a negative number
  author 	VARCHAR(70) 	NOT NULL,
  price 	NUMERIC(4,2) 	NOT NULL	CHECK (price > 0), -- checks the price is a positive number
  inventoryid 	VARCHAR(6) 	NOT NULL,
  genre VARCHAR(20) DEFAULT 'Fantasy',
  PRIMARY KEY (bookid),
  FOREIGN KEY (inventoryid) REFERENCES inventory(inventoryid) ON DELETE CASCADE, -- remove record if referenced inventoryid is deleted
  UNIQUE (isbn));

CREATE TABLE orderitem
(
  orderitemid 	VARCHAR(6) 	NOT NULL,
  orderid 	VARCHAR(6) 	NOT NULL,
  bookid 	VARCHAR(6) 	NOT NULL,
  quantity 	INT 		NOT NULL,
  PRIMARY KEY (orderitemid),
  FOREIGN KEY (orderid) REFERENCES orders(orderid), -- implied that this is a ON DELETE RESTRICT
  FOREIGN KEY (bookid) REFERENCES book(bookid) ON DELETE CASCADE -- remove record if referenced bookid is deleted
);

CREATE TABLE sales
(
  salesid 	VARCHAR(6) 	NOT NULL,
  sdate 	DATE 		NOT NULL,
  typecard 	VARCHAR(20) 	NOT NULL,
  cardnumber 	VARCHAR(20) 	NOT NULL,
  orderid 	VARCHAR(6) 	NOT NULL,
  code 		VARCHAR(15),
  PRIMARY KEY (salesid),
  FOREIGN KEY (orderid) REFERENCES orders(orderid), -- implied that this is a ON DELETE RESTRICT
  FOREIGN KEY (code) REFERENCES discount(code) -- implied that this is a ON DELETE RESTRICT
);

CREATE TABLE BooksDim
(
  BookKey INT NOT NULL,
  BookId VARCHAR(6) NOT NULL,  
  Title VARCHAR(255) NOT NULL,
  Author VARCHAR(100) NOT NULL,
  StockQuantity INT NOT NULL,
  Genre VARCHAR(100) NOT NULL,
  Price NUMERIC(5,2) NOT NULL,
  PRIMARY KEY (BookKey)
);

CREATE TABLE EmployeeDim
(
  EmployeeKey INT NOT NULL,
  EmployeeID INT NOT NULL,
  Role VARCHAR(80) Default 'Sales Associate',
  EmpName VARCHAR(100) NOT NULL,
  phoneNumber VARCHAR(20) NOT NULL,
  PRIMARY KEY (EmployeeKey)
);

CREATE TABLE CustomerDim
(
  CustomerKey INT NOT NULL,
  CustomerID VARCHAR(5) NOT NULL,
  Cname VARCHAR(100) NOT NULL,
  Email VARCHAR(255) NOT NULL,
  PhoneNumber VARCHAR(20) NOT NULL,
  PRIMARY KEY (CustomerKey)
);

CREATE TABLE CalendarDim
(
  CalendarKey INT NOT NULL,
  FullDate DATE NOT NULL,
  DayOfWeek VARCHAR(100) NOT NULL,
  DayOfMonth INT NOT NULL,
  Month VARCHAR(100) NOT NULL,
  Year INT NOT NULL,
  PRIMARY KEY (CalendarKey)
);

CREATE TABLE OrdersDim
(
  OrdersKey INT NOT NULL,
  OrderID VARCHAR(6) NOT NULL,
  orderDate DATE NOT NULL,
  customerID VARCHAR(5) NOT NULL,
  employeeID INT NOT NULL,
  PRIMARY KEY (OrdersKey)
);

CREATE TABLE OrderItemDim
(
  OrderItemKey INT NOT NULL,
  OrderItemID VARCHAR(6) NOT NULL,
  OrderID VARCHAR(6) NOT NULL,
  BookID VARCHAR(6) NOT NULL,
  Quantity INT NOT NULL,
  PRIMARY KEY (OrderItemKey)
);

CREATE TABLE SalesFact
(
  salesfactKey INT NOT NULL,  
  BookKey INT NOT NULL,
  EmployeeKey INT NOT NULL,
  CustomerKey INT NOT NULL,
  CalendarKey INT NOT NULL,
  OrdersKey INT NOT NULL,
  OrderItemKey INT NOT NULL,
  TotalAmount FLOAT NOT NULL,
  TotalUnitsSold INT NOT NULL,
  PRIMARY KEY (salesfactKey),  
  FOREIGN KEY (BookKey) REFERENCES BooksDim(BookKey),
  FOREIGN KEY (EmployeeKey) REFERENCES EmployeeDim(EmployeeKey),
  FOREIGN KEY (CustomerKey) REFERENCES CustomerDim(CustomerKey),
  FOREIGN KEY (CalendarKey) REFERENCES CalendarDim(CalendarKey),
  FOREIGN KEY (OrdersKey) REFERENCES OrdersDim(OrdersKey),
  FOREIGN KEY (OrderItemKey) REFERENCES OrderItemDim(OrderItemKey)
);

-- Calendar
CREATE SEQUENCE calKey_Seq
START WITH 1
INCREMENT BY  1; 

DECLARE
    
    CURSOR orders_c IS
       SELECT DISTINCT oDate
       FROM orders;
       
    cal_rec orders_c%rowtype;
    
BEGIN
    OPEN orders_c;
    
    LOOP
        FETCH orders_c INTO cal_rec;
        EXIT WHEN orders_c%notfound;
        
        INSERT INTO CalendarDim 
            VALUES(calKey_Seq.nextval, cal_rec.oDate, 
                  TO_CHAR(cal_rec.oDate, 'DAY'),
                   EXTRACT(DAY FROM cal_rec.oDate),
                  TO_CHAR(cal_rec.oDate, 'MONTH'),
                  EXTRACT(YEAR FROM cal_rec.oDate));
   END LOOP;
   CLOSE orders_c;
END;
/

--SalesFactTable
CREATE SEQUENCE sftKey_Seq
START WITH 1
INCREMENT BY  1; 

DECLARE
    
    CURSOR sales_c IS 
        SELECT b.bookKey, e.EmployeeKey, c.CustomerKey, cal.CalendarKey, o.OrdersKey, oi.OrderItemKey,
                   SUM(b.price * oi.quantity) AS totalAmount, SUM(oi.quantity) AS totalUnitsSold
        FROM booksDim b, 
             employeeDim e, 
             customerDim c, 
             calendarDim cal, 
             ordersDim o, 
             orderitemDim oi          
        WHERE oi.bookid = b.bookid AND 
              o.customerid = c.customerid AND 
              o.orderid = oi.orderid AND 
              o.employeeid = e.employeeid AND 
              o.orderdate = cal.fullDate
        GROUP BY b.bookKey, e.employeeKey, c.CustomerKey, cal.CalendarKey, o.OrdersKey, oi.OrderItemKey;
                   
    sales_rec sales_c%rowtype;               

BEGIN
     OPEN sales_c;              
    
     LOOP              
        FETCH sales_c INTO sales_rec;
        EXIT WHEN sales_c%notfound;
     INSERT INTO salesFact 
        VALUES (sftKey_Seq.nextval, sales_rec.bookKey, sales_rec.employeeKey, sales_rec.customerKey, sales_rec.CalendarKey,
               sales_rec.OrdersKey, sales_rec.OrderItemKey, sales_rec.totalAmount, sales_rec.totalUnitsSold);
    END LOOP;
    CLOSE sales_c;               
END;
/

-- For the tables on the Books 2024 page. The purpose is to show the pivot tables

SELECT * FROM 
(
    SELECT b.author AS "Author Name", 
           b.title AS "Book Title", 
           SUM(s.totalUnitsSold) AS "Total Sold", 
           SUM(s.totalAmount) AS "Total Revenue" 
    FROM booksdim b, salesfact s, orderitemdim oi, ordersdim o, calendardim c 
    WHERE s.bookKey = b.bookKey AND 
          s.orderitemKey = oi.orderitemKey AND 
          o.orderdate = c.fulldate AND 
          c.dayofWeek = 'WEDNESDAY' AND 
          b.genre = 'Fantasy'
    GROUP BY b.author, b.title
)
PIVOT 
(
   SUM("Total Sold") AS "Total Sold", 
   SUM("Total Revenue") AS "Total Revenue"
   FOR "Author Name" IN ('Emily Cartwright' AS "Emily Cartwright",
                         'Sarah Kensington' AS "Sarah Kensington",
                         'Elsa Iceland' AS "Elsa Iceland",
                         'Jack Stiff' AS "Jack Stiff",
                        'Kristen Kurt' AS "Kristen Kurt",
                        'Natalie Summers' AS "Natalie Summers")
);

)

   FOR Genre IN ('Fantasy' AS Fantasy, 'Science' AS Science)
);

--FOR DELETE

CREATE TABLE orders_backup
(
    orderid INT NOT NULL,
    odate DATE NOT NULL,
    customerid VARCHAR(5) NOT NULL,
    employeeid VARCHAR(5) DEFAULT '1',
    timeExp TIMESTAMP, 
    PRIMARY KEY (orderid)
);

--from docs.oracle.com (formatting SQL * PLUS Reports)
COLUMN timeExp FORMAT A29;

-- BEFORE DELETE trigger
CREATE OR REPLACE TRIGGER BEFORE_DELETE
BEFORE DELETE ON orders
FOR EACH ROW

BEGIN
INSERT INTO orders_backup (orderid, odate, customerid, employeeid,  timeExp)
  VALUES (:OLD.orderid, :OLD.odate, :OLD.customerid, :OLD.employeeid, CURRENT_TIMESTAMP);
END;
/

--Displays how this works in action. Beware, this will delete from the Orders table
DELETE FROM orders WHERE employeeid = 1;
SELECT * FROM orders;
SELECT * FROM orders_backup;
