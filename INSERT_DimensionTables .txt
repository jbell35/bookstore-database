-- Calendar Dimension
CREATE SEQUENCE calKey_Seq
START WITH 1
INCREMENT BY  1; 

DECLARE
    
    CURSOR orders_c IS
       SELECT DISTINCT oDate
       FROM orders;
       
    cal_rec orders_c%rowtype;
    
BEGIN
    OPEN orders_c;
    
    LOOP
        FETCH orders_c INTO cal_rec;
        EXIT WHEN orders_c%notfound;
        
        INSERT INTO CalendarDim 
            VALUES(calKey_Seq.nextval, cal_rec.oDate, 
                  TO_CHAR(cal_rec.oDate, 'DAY'),
                   EXTRACT(DAY FROM cal_rec.oDate),
                  TO_CHAR(cal_rec.oDate, 'MONTH'),
                  EXTRACT(YEAR FROM cal_rec.oDate));
   END LOOP;
   CLOSE orders_c;
END;
/


--Orders Dimension

CREATE SEQUENCE oKey_Seq
START WITH 1
INCREMENT BY  1; 

DECLARE
    CURSOR order_c IS 
        SELECT orderid, odate, customerid, employeeid
        FROM orders;
    order_rec order_c%rowtype;               
                   
BEGIN
    OPEN order_c;
    
     LOOP              
         FETCH order_c INTO order_rec;
         EXIT WHEN order_c%notfound;
         
         INSERT INTO OrdersDim
             VALUES (oKey_Seq.nextval, order_rec.orderid, order_rec.odate, 
                    order_rec.customerid, order_rec.employeeid, 
                    (SELECT SUM(b.price * oi.quantity)
                     FROM orderitem oi
                     JOIN book b ON oi.bookid = b.bookid
                     WHERE oi.orderid = order_rec.orderid));
    END LOOP;
    CLOSE order_c;               
END;
/

--SalesFactTable Dimension
CREATE SEQUENCE sftKey_Seq
START WITH 1
INCREMENT BY  1; 

DECLARE
    
    CURSOR sales_c IS 
        SELECT b.bookKey, e.EmployeeKey, c.CustomerKey, cal.CalendarKey, o.OrdersKey, oi.OrderItemKey,
                   SUM(b.price * oi.quantity) AS totalAmount, SUM(oi.quantity) AS totalUnitsSold
        FROM booksDim b, 
             employeeDim e, 
             customerDim c, 
             calendarDim cal, 
             ordersDim o, 
             orderitemDim oi          
        WHERE oi.bookid = b.bookid AND 
              o.customerid = c.customerid AND 
              o.orderid = oi.orderid AND 
              o.employeeid = e.employeeid AND 
              o.orderdate = cal.fullDate
        GROUP BY b.bookKey, e.employeeKey, c.CustomerKey, cal.CalendarKey, o.OrdersKey, oi.OrderItemKey;
                   
    sales_rec sales_c%rowtype;               

BEGIN
     OPEN sales_c;              
    
     LOOP              
        FETCH sales_c INTO sales_rec;
        EXIT WHEN sales_c%notfound;
     INSERT INTO salesFact 
        VALUES (sftKey_Seq.nextval, sales_rec.bookKey, sales_rec.employeeKey, sales_rec.customerKey, sales_rec.CalendarKey,
               sales_rec.OrdersKey, sales_rec.OrderItemKey, sales_rec.totalAmount, sales_rec.totalUnitsSold);
    END LOOP;
    CLOSE sales_c;               
END;
